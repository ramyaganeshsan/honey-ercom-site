GET PRODUCT DETAIL
    SELECT 
        product.deal_title,
        product.deal_id,
        product.deal_description, 
        product.deal_value, 
        product.deal_price, 
        product.delivery_period,
        product.category_id,
        product.sub_category_id,
        product.sec_category_id,
        CASE WHEN product.user_limit_quantity > 0 THEN true ELSE false END AS inStock,
        AVG(rate_review.rating) as ratings, 
        COUNT(rate_review.id) as total_reviews,
        COUNT(cart_items.deal_id) as in_cart,
        main_category.category_name as main_category_name,
        sub_category.category_name as sub_category_name,
        second_level_category.category_name as second_level_category_name
    FROM product 
    LEFT JOIN rate_review ON rate_review.type_id = product.deal_id AND rate_review.approve_status = 1
    LEFT JOIN category AS main_category ON main_category.category_id = product.category_id
    LEFT JOIN category AS sub_category ON sub_category.category_id = product.sub_category_id
    LEFT JOIN category AS second_level_category ON second_level_category.category_id = product.sec_category_id
    LEFT JOIN ( SELECT deal_id FROM cart_items JOIN cart ON cart_items.cart_id = cart.cart_id WHERE cart.cart_transaction_status != 1 ) AS cart_items ON cart_items.deal_id = product.deal_id
    WHERE product.deal_status = 1 AND product.deal_key = "yyN3Nl4A";

GET CART PRODUCTS
    SELECT 
        cart_items.deal_id,
        cart_items.deal_key,
        cart_items.deal_value,
        cart_items.item_quantity,
        products.deal_value as currentPrice,
        products.deal_title
    FROM cart
    LEFT JOIN cart_items ON cart_items.cart_id = cart.cart_id
    LEFT JOIN ( SELECT deal_id, deal_title, deal_value FROM product WHERE deal_status = 1 ) AS products ON products.deal_id = cart_items.deal_id
    WHERE cart.user_id = 8719 AND cart.cart_transaction_status != 1
    GROUP BY cart_items.item_id 
    ORDER BY cart_items.item_id DESC;

GET CHECKOUT COUNTRY DETAILS 
    SELECT * FROM state WHERE state_country_id = 139;
    UPDATE state SET statestatus = 1 WHERE state_country_id = 139;

    SELECT * FROM city WHERE country_id = 139;
    UPDATE state SET statestatus = 1 WHERE state_country_id = 139;

    SELECT 
        country_name,
        country_status,
        concat('[', 
            GROUP_CONCAT(
                concat(
                    '{state_id:',state.state_id,','
                    'state_name:','"',state.state_name,'"','}'
                )
            ), 
        ']') as states,
        concat('[', 
            GROUP_CONCAT(
                concat(
                    '{',
                        'city_id:',city.city_id,',',
                        'country_id:',city.country_id,',',
                        'state_id:','"',city.stateid,'"',
                        'city_name:','"',city.city_name,'"',
                        'delivery_charge:','"',city.delivery_charge,'"'
                    '}'
                )
            ), 
        ']') as cities
    FROM country
    LEFT JOIN state ON state.state_country_id = country.country_id AND state.statestatus = 1
    LEFT JOIN city ON city.country_id = country.country_id AND city.city_status = 1
    WHERE country_status = 1;
